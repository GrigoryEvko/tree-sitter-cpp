A conversion function defined outside its class after a specifier or a macro
================================================================================

// The name of a conversion function defined outside its class is a qualified conversion-function-id
// ([class.conv.fct], [class.mfct] p2). GCC reads the keyword `operator` and then a
// conversion-type-id in `cp_parser_conversion_function_id` (gcc/cp/parser.cc:19612), and Clang in
// `Parser::ParseUnqualifiedId` (clang/lib/Parse/ParseExprCXX.cpp:2785). The keyword is never a
// name. The specifier or the macro before the qualified name splits the parse before `n`, and the
// keyword has no action after `n::X::` in the version of a declaration. Before this test, the lexer
// gave the word token there, and the fork read `n::X::operator` as the type of the first three
// definitions and the conversion-type-id as the name of a function, with no error node. The fourth
// definition has no specifier, and its tree is the same before and after the test. Refer to
// `QUALIFIED_TYPE_NAME_RESERVED_WORDS`.
struct T { };
template <class U> struct Vec { };
namespace std { struct string { }; }
namespace n { struct X { operator T() const; operator Vec<T>() const; constexpr operator std::string() const; }; }
template <class U> struct R { operator U() noexcept; };
inline n::X::operator T() const LIFETIME_BOUND { return T(); }
constexpr n::X::operator std::string() const { return std::string(); }
template <class U> SIMDJSON_INLINE R<U>::operator U() noexcept { return U(); }
n::X::operator Vec<T>() const { return Vec<T>(); }
